DROP PROCEDURE IF EXISTS `sp_SaveSale`;
CREATE DEFINER=`DFLPOS`@`%` PROCEDURE `sp_SaveSale`(IN _saleid BIGINT, IN _saledate DATETIME, IN _saletotal DECIMAL(19,2), IN _saletax DECIMAL(19,2), 
	IN _salesubtotal DECIMAL(19,2), IN _salediscount DECIMAL(19,2), IN _totalcost DECIMAL(19,2), IN _customerid BIGINT, IN _userid INT,
    IN _currencyrate DECIMAL(10,2), IN _status CHAR(1), IN _customercard BIGINT, IN _posid VARCHAR(6), IN _sitecode VARCHAR(6),IN _tribu_customer VARCHAR(10))
BEGIN
	DECLARE _returnvalue BIGINT;
    DECLARE _count INT;
    SET _returnvalue = 0;
    -- VERIFICAMOS QUE LA VENTA NO EXISTA YA
    IF(SELECT 1 = 1 FROM `dflpos`.`sale` WHERE `saleid`=_saleid) THEN
		BEGIN
			SET _returnvalue = -1;
		END;
	ELSE
		BEGIN
			INSERT INTO `dflpos`.`sale` (`saleid`,`saledate`,`saletotal`,`saletax`,`salesubtotal`,`salediscount`,`totalcost`,
									`customerid`,`userid`,`CurrencyRate`,`status`,`customercard`,`posid`,`sitecode`,`tribu_customer`)
			VALUES	
					(_saleid, _saledate, _saletotal, _saletax, _salesubtotal, _salediscount, _totalcost, _customerid, _userid, _currencyrate, _status,
					_customercard, _posid, _sitecode,_tribu_customer);
			SET _returnvalue = _saleid;
        END;
	END IF;
    SELECT _returnvalue AS ReturnValue;
END;

CREATE DEFINER=`DFLPOS`@`%` PROCEDURE `sp_AddSaleDetail`(IN _saleid BIGINT, IN _item_id VARCHAR(20), IN _quantity DECIMAL(19,2), IN _price DECIMAL(19,2), 
		IN _tax DECIMAL(19,2),IN _tax_percent DECIMAL(19,2), IN _total DECIMAL(19,2), IN _totaldiscount DECIMAL(19,2),IN _barcode VARCHAR(50), 
        IN _cost DECIMAL(19,2),IN _lastcost DECIMAL(19,2), IN _status CHAR(1), IN _transid INT, IN _priceoriginal DECIMAL(19,2), IN _tribu_promo CHAR(1))
BEGIN
	
	IF(SELECT 1 = 1 FROM `dflpos`.`saledetail` WHERE `saleid`=_saleid AND `item_id`=_item_id AND `transid`=_transid) THEN
		BEGIN
			SELECT 1 AS ReturnValue;
		END;
	ELSE
		BEGIN
			INSERT INTO `dflpos`.`saledetail`(`saleid`,`item_id`,`quantity`,`price`,`tax`,`tax_percent`,`total`,`totaldiscount`,`barcode`,`cost`,
			`lastcost`,`status`,`transid`,`priceoriginal`,`tribu_promo`)
			VALUES (_Saleid, _item_id, _quantity, _price, _tax, _tax_percent, _total, _totaldiscount, _barcode, _cost, _lastcost, _status,_transid,_priceoriginal,_tribu_promo);
			SELECT 1 AS ReturnValue;
		END;
	END IF;
